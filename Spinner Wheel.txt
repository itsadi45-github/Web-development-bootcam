Java-Script Files:

var spinning = false;

function spinWheel() {
    if (!spinning) {
        spinning = true;
        var spinner = document.getElementById("spinner");
        var resultText = document.getElementById("result");

        // Generate a random degree for spinning
        var randomDegree = Math.floor(Math.random() * (360 - 180) + 180);

        // Apply the rotation animation
        spinner.style.transition = "transform 3s ease-out";
        spinner.style.transform = "rotate(" + randomDegree + "deg)";

        // Display the result after spinning
        setTimeout(function () {
            spinning = false;
            spinner.style.transition = "none";
            spinner.style.transform = "rotate(0deg)";

            // Determine the discount based on the degree
            var discount = calculateDiscount(randomDegree);
            resultText.innerHTML = "Congratulations! You've won a " + discount + "% discount.";
        }, 3000);
    }
}

function calculateDiscount(degrees) {
    // You can customize the discount levels based on the spinner's degree
    if (degrees >= 0 && degrees < 60) {
        return 10;
    } else if (degrees >= 60 && degrees < 120) {
        return 20;
    } else if (degrees >= 120 && degrees < 180) {
        return 30;
    } else if (degrees >= 180 && degrees < 240) {
        return 40;
    } else if (degrees >= 240 && degrees < 300) {
        return 50;
    } else {
        return 0;
    }
}
